local function isAlive(plr)
    if not plr then return false end
    local char = plr.Character
    if not char then return false end
    if not char:IsDescendantOf(Workspace) then return false end

    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not humanoid then return false end
    if humanoid.Health <= 0 then return false end

    local state = humanoid:GetState()
    if state == Enum.HumanoidStateType.Dead then return false end
    if state == Enum.HumanoidStateType.Physics then return false end

    local rootPart = char:FindFirstChild("HumanoidRootPart")
    if not rootPart then return false end
    if rootPart.Position.Y < -300 then return false end
    if rootPart.Anchored then return false end
    if humanoid.PlatformStand then return false end

    local head = char:FindFirstChild("Head")
    if head and head.Transparency >= 1 then return false end
    if char:FindFirstChild("DeathTag") or char:FindFirstChild("IsDead") then return false end

    local creatorTag = humanoid:FindFirstChild("creator")
    if creatorTag then return false end

    return true
end

-- Deep scan for armor/shield values on a character
local function GetArmorData(plr)
    local char = plr.Character
    if not char then return 0, 100 end

    local armorObj = nil
    local armorNames = {"Armor", "Shield", "Vest", "BodyArmor", "armor", "shield", "vest"}

    -- Check direct children of character
    for _, name in ipairs(armorNames) do
        armorObj = char:FindFirstChild(name)
        if armorObj then break end
    end

    -- Check leaderstats
    if not armorObj then
        local ls = plr:FindFirstChild("leaderstats")
        if ls then
            for _, name in ipairs(armorNames) do
                armorObj = ls:FindFirstChild(name)
                if armorObj then break end
            end
        end
    end

    -- Check inside humanoid
    if not armorObj then
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if humanoid then
            for _, name in ipairs(armorNames) do
                armorObj = humanoid:FindFirstChild(name)
                if armorObj then break end
            end
        end
    end

    -- Deep scan removed for performance reasons.

    local armorValue = 0
    local maxArmor = 100

    if armorObj and (armorObj:IsA("NumberValue") or armorObj:IsA("IntValue")) then
        armorValue = armorObj.Value
        -- Try to find max armor
        local maxObj = char:FindFirstChild("MaxArmor") or char:FindFirstChild("MaxShield") or char:FindFirstChild("maxArmor")
        if maxObj and (maxObj:IsA("NumberValue") or maxObj:IsA("IntValue")) and maxObj.Value > 0 then
            maxArmor = maxObj.Value
        end
    end

    return armorValue, maxArmor
end

local function ClearOpStats()
    for _, conn in ipairs(OpStatsConnections) do
        if conn then conn:Disconnect() end
    end
    OpStatsConnections = {}
end

local function ApplyOpStats(char)
    ClearOpStats()
    if not OpStatsEnabled then return end
    local humanoid = char:WaitForChild("Humanoid", 5)
    if not humanoid then return end
    humanoid.MaxHealth = 100
    humanoid.Health = humanoid.MaxHealth
    humanoid.WalkSpeed = 45
    local c1 = humanoid:GetPropertyChangedSignal("Health"):Connect(function()
        if OpStatsEnabled and humanoid.Health < humanoid.MaxHealth then
            humanoid.Health = humanoid.MaxHealth
        end
    end)
    local c2 = humanoid:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
        if OpStatsEnabled and humanoid.WalkSpeed ~= 45 then
            humanoid.WalkSpeed = 45
        end
    end)
    table.insert(OpStatsConnections, c1)
    table.insert(OpStatsConnections, c2)
end

--=============================================
-- Invisibility (Ghost Mode)
--=============================================
local isGhosting = false
local RealCharacter = nil
local FakeCharacter = nil

local function EnableInvisibility(state)
    InvisibleEnabled = state
    if state then
        if isGhosting then return end
        RealCharacter = LocalPlayer.Character
        if not RealCharacter then return end
        local realHRP = RealCharacter:FindFirstChild("HumanoidRootPart")
        local realHum = RealCharacter:FindFirstChild("Humanoid")
        if not realHRP or not realHum then return end

        isGhosting = true
        RealCharacter.Archivable = true
        FakeCharacter = RealCharacter:Clone()
        FakeCharacter.Name = LocalPlayer.Name
        FakeCharacter.Parent = Workspace

        for _, part in ipairs(FakeCharacter:GetDescendants()) do
            if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                part.Transparency = 1
            elseif part:IsA("Decal") or part:IsA("Texture") then
                part.Transparency = 1
            elseif part:IsA("BillboardGui") then
                part.Enabled = false
            end
        end

        FakeCharacter:FindFirstChild("HumanoidRootPart").CFrame = realHRP.CFrame
        LocalPlayer.Character = FakeCharacter
        Camera.CameraSubject = FakeCharacter:FindFirstChild("Humanoid")

        local anim = FakeCharacter:FindFirstChild("Animate")
        if anim then anim.Disabled = true; task.wait(); anim.Disabled = false end

        RunService:BindToRenderStep("InvisibleHold", 100, function()
            if isGhosting and RealCharacter and FakeCharacter then
                local rRoot = RealCharacter:FindFirstChild("HumanoidRootPart")
                local fRoot = FakeCharacter:FindFirstChild("HumanoidRootPart")
                if rRoot and fRoot then
                    rRoot.Velocity = Vector3.new(0, 0, 0)
                    rRoot.CFrame = fRoot.CFrame * CFrame.new(0, -50, 0)
                end
            end
        end)
    else
        if not isGhosting then return end
        isGhosting = false
        RunService:UnbindFromRenderStep("InvisibleHold")

        if RealCharacter then
            local realHRP = RealCharacter:FindFirstChild("HumanoidRootPart")
            local fakeHRP = FakeCharacter and FakeCharacter:FindFirstChild("HumanoidRootPart")
            if realHRP then
                realHRP.Anchored = false
                if fakeHRP then realHRP.CFrame = fakeHRP.CFrame + Vector3.new(0, 5, 0) end
            end
            LocalPlayer.Character = RealCharacter
            Camera.CameraSubject = RealCharacter:FindFirstChild("Humanoid")
            if FakeCharacter then FakeCharacter:Destroy(); FakeCharacter = nil end
            local anim = RealCharacter:FindFirstChild("Animate")
            if anim then anim.Disabled = true; task.wait(); anim.Disabled = false end
        end
    end
end

--=============================================
-- Character Tracking
--=============================================
local function UpdateCharacter(char)
    if isGhosting and char == FakeCharacter then
        Character = char
        HRP = char:WaitForChild("HumanoidRootPart", 10)
        return
    end
    isGhosting = false
    RunService:UnbindFromRenderStep("InvisibleHold")
    Character = char
    HRP = char:WaitForChild("HumanoidRootPart", 10)
    if InvisibleEnabled then
        task.spawn(function() task.wait(1); EnableInvisibility(true) end)
    end
    if OpStatsEnabled then
        task.spawn(function() ApplyOpStats(char) end)
    end
end

if Character then UpdateCharacter(Character) end
LocalPlayer.CharacterAdded:Connect(UpdateCharacter)

            end
        end)

            if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                local delta = input.Position - dragStart
                gui.Position = UDim2.new(
                    startPos.X.Scale, startPos.X.Offset + delta.X,
                    startPos.Y.Scale, startPos.Y.Offset + delta.Y
                )
            end
        end)

            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then

    local function SkidFling(TargetPlayer)
        local Char = LocalPlayer.Character
        local Humanoid = Char and Char:FindFirstChildOfClass("Humanoid")
        local RootPart = Humanoid and Humanoid.RootPart
        local TCharacter = TargetPlayer.Character
        if not TCharacter then return end
        local THumanoid = TCharacter:FindFirstChildOfClass("Humanoid")
        local TRootPart = THumanoid and THumanoid.RootPart
        local THead = TCharacter:FindFirstChild("Head")
        local Accessory = TCharacter:FindFirstChildOfClass("Accessory")
        local Handle = Accessory and Accessory:FindFirstChild("Handle")

        if not (Char and Humanoid and RootPart) then return end
        if RootPart.Velocity.Magnitude < 50 then getgenv().OldPos = RootPart.CFrame end
        if THumanoid and THumanoid.Sit then return end

        if THead then Camera.CameraSubject = THead
        elseif Handle then Camera.CameraSubject = Handle
        elseif THumanoid and TRootPart then Camera.CameraSubject = THumanoid end

        local FPos = function(BasePart, Pos, Ang)
            RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
            Char:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
            RootPart.Velocity = Vector3.new(9e7, 9e7 * 10, 9e7)
            RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
        end

        local SFBasePart = function(BasePart)
            local TimeToWait = 2
            local Time = tick()
            local Angle = 0
            repeat
                if RootPart and THumanoid then
                    if BasePart.Velocity.Magnitude < 50 then
                        Angle = Angle + 100
                        FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0)); task.wait()
                        FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0)); task.wait()
                        FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle), 0, 0)); task.wait()
                        FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle), 0, 0)); task.wait()
                    else
                        FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0)); task.wait()
                        FPos(BasePart, CFrame.new(0, -1.5, -THumanoid.WalkSpeed), CFrame.Angles(0, 0, 0)); task.wait()
                        FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0)); task.wait()
                        FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0)); task.wait()
                    end
                end
            until Time + TimeToWait < tick() or not FlingActive
        end

        Workspace.FallenPartsDestroyHeight = 0 / 0
        local BV = Instance.new("BodyVelocity")
        BV.Parent = RootPart
        BV.Velocity = Vector3.new(0, 0, 0)
        BV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

        if TRootPart then SFBasePart(TRootPart)
        elseif THead then SFBasePart(THead)
        elseif Handle then SFBasePart(Handle) end

        BV:Destroy()
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
        Camera.CameraSubject = Humanoid

        if getgenv().OldPos then
            repeat
                RootPart.CFrame = getgenv().OldPos * CFrame.new(0, 0.5, 0)
                Char:SetPrimaryPartCFrame(getgenv().OldPos * CFrame.new(0, 0.5, 0))
                Humanoid:ChangeState("GettingUp")
                for _, part in pairs(Char:GetChildren()) do
                    if part:IsA("BasePart") then
                        part.Velocity = Vector3.new()
                        part.RotVelocity = Vector3.new()
                    end
                end
                task.wait()
            until (RootPart.Position - getgenv().OldPos.p).Magnitude < 25
            Workspace.FallenPartsDestroyHeight = getgenv().FPDH
        end
    end

    local function GetSilentAimTarget()
        local Closest = nil
        local DistanceToMouse = CrosshairRadius
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and isAlive(plr) then
                local Char = plr.Character
                local targetPart
                if SilentAimTargetPart == "Random" then
                    targetPart = Char:FindFirstChild(ValidTargetParts[math.random(1, #ValidTargetParts)])
                else
                    targetPart = Char:FindFirstChild(SilentAimTargetPart)
                end
                if not targetPart then continue end
                local screenPos, onViewport = Camera:WorldToViewportPoint(targetPart.Position)
                if onViewport then
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - UserInputService:GetMouseLocation()).Magnitude
                    if dist <= DistanceToMouse then
                        Closest = targetPart
                        DistanceToMouse = dist
                    end
                end
            end
        end
        return Closest
    end

    --=============================================
    -- Green Box Detection (teammate indicator skip)
    --=============================================
    local greenBoxCache = {}
    local function hasGreenBox(char)
        if not char then return false end
        if greenBoxCache[char] and (tick() - greenBoxCache[char].lastCheck < 2) then
            return greenBoxCache[char].hasGreen
        end

        local hasGreen = false
        local root = char:FindFirstChild("HumanoidRootPart")
        local head = char:FindFirstChild("Head")

        local itemsToCheck = {}
        for _, c in pairs(char:GetChildren()) do table.insert(itemsToCheck, c) end
        if head then
            for _, c in pairs(head:GetChildren()) do table.insert(itemsToCheck, c) end
        end

        for _, v in pairs(itemsToCheck) do
            if v:IsA("BillboardGui") or v:IsA("SurfaceGui") then
                for _, ui in pairs(v:GetDescendants()) do
                    if ui:IsA("Frame") and ui.BackgroundColor3.G > 0.5 and ui.BackgroundColor3.R < 0.3 and ui.BackgroundColor3.B < 0.3 then
                        hasGreen = true
                        break
                    end
                end
            elseif v:IsA("BasePart") and v.Name ~= "HumanoidRootPart" and v.Color.G > 0.5 and v.Color.R < 0.3 and v.Color.B < 0.3 then
                if root and v.Position.Y > root.Position.Y then
                    hasGreen = true
                end
            end
            if hasGreen then break end
        end

        greenBoxCache[char] = {hasGreen = hasGreen, lastCheck = tick() + (math.random() * 0.5)}
        return hasGreen
    end

    --=============================================
    -- ESP + Aimbot Update Loop
    --=============================================
    local function UpdateESP_And_Aimbot(dt)
        -- Clean up disconnected players
        for plr, esp in pairs(ESP_Cache) do
            if not plr or not plr.Parent then
                ClearESP(plr)
            end
        end

        if not FOVCircle then return end
        FOVCircle.Position = UserInputService:GetMouseLocation()
        FOVCircle.Radius = CrosshairRadius
        FOVCircle.Visible = CrosshairEnabled

        local closestDistance = CrosshairRadius
        local targetHead = nil

        for _, plr in ipairs(Players:GetPlayers()) do
            if plr == LocalPlayer then continue end

            if not ESP_Cache[plr] then CreateESP(plr) end
            local esp = ESP_Cache[plr]
            local hrp = getHRP(plr)
            local head = getHead(plr)
            local dist = hrp and (hrp.Position - Camera.CFrame.Position).Magnitude or 0
            local hideEsp = hasGreenBox(plr.Character)
            local playerIsAlive = isAlive(plr)

            -- ESP rendering
            if WallhackEnabled and hrp and head and dist <= ESPMaxDistance and not hideEsp and playerIsAlive then
                local rootPos, onViewport = Camera:WorldToViewportPoint(hrp.Position)
                local headPos = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
                local legPos = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))

                if onViewport then
                    local color = Color3.fromRGB(255, 255, 255)
                    local height = math.abs(headPos.Y - legPos.Y)
                    local width = height / 2

                    -- Name + Distance
                    esp.Box.Visible = false
                    esp.Name.Position = Vector2.new(rootPos.X, headPos.Y - 20)
                    esp.Name.Text = plr.Name .. " [" .. math.floor(dist) .. "m]"
                    esp.Name.Color = color
                    esp.Name.Visible = ShowNames

                    -- Tracer line
                    esp.Tracer.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                    esp.Tracer.To = Vector2.new(rootPos.X, rootPos.Y + height / 2)
                    esp.Tracer.Color = color
                    esp.Tracer.Visible = ShowLines

                    -- ========================================
                    -- HEALTH BAR (RED) — Vertical, Left Side
                    -- ========================================
                    local humanoid = plr.Character and plr.Character:FindFirstChildOfClass("Humanoid")
                    local currentHP = humanoid and humanoid.Health or 0
                    local maxHP = humanoid and humanoid.MaxHealth or 100
                    if maxHP <= 0 then maxHP = 100 end
                    local healthPercent = math.clamp(currentHP / maxHP, 0, 1)

                    -- Armor detection
                    local armorValue, maxArmor = GetArmorData(plr)
                    local armorPercent = math.clamp(armorValue / maxArmor, 0, 1)
                    local hasArmor = armorValue > 0

                    -- Bar positioning
                    local barW = 3
                    local barGap = 3
                    local boxLeft = rootPos.X - width / 2
                    local boxTop = headPos.Y
                    local boxBottom = legPos.Y
                    local barH = math.max(math.abs(boxBottom - boxTop), 12)

                    -- Health bar position (just left of bounding box)
                    local hpX = boxLeft - barGap - barW

                    -- Health bar background (full height, black)
                    esp.HealthBarBG.Position = Vector2.new(hpX - 1, boxTop - 1)
                    esp.HealthBarBG.Size = Vector2.new(barW + 2, barH + 2)
                    esp.HealthBarBG.Visible = ShowHealthBars

                    -- Health bar fill (RED, bottom-up)
                    local hpFill = barH * healthPercent
                    esp.HealthBar.Position = Vector2.new(hpX, boxTop + (barH - hpFill))
                    esp.HealthBar.Size = Vector2.new(barW, hpFill)
                    -- Bright red at full HP, dark red at low HP
                    local redVal = math.floor(80 + 175 * healthPercent)
                    esp.HealthBar.Color = Color3.fromRGB(redVal, math.floor(20 * healthPercent), math.floor(20 * healthPercent))
                    esp.HealthBar.Visible = ShowHealthBars

                    -- Health number text (positioned at the fill top)
                    esp.HealthText.Position = Vector2.new(hpX - 2, boxTop + (barH - hpFill) - 14)
                    esp.HealthText.Text = tostring(math.floor(currentHP)) .. "/" .. tostring(math.floor(maxHP))
                    esp.HealthText.Color = Color3.fromRGB(255, 100, 100)
                    esp.HealthText.Visible = ShowHealthBars

                    -- ========================================
                    -- ARMOR BAR (BLUE) — Left of Health Bar
                    -- ========================================
                    local armorX = hpX - barGap - barW - 2

                    -- Armor bar background
                    esp.ArmorBarBG.Position = Vector2.new(armorX - 1, boxTop - 1)
                    esp.ArmorBarBG.Size = Vector2.new(barW + 2, barH + 2)
                    esp.ArmorBarBG.Visible = ShowHealthBars

                    -- Armor bar fill (BLUE, bottom-up)
                    local armorFill = barH * armorPercent
                    esp.ArmorBar.Position = Vector2.new(armorX, boxTop + (barH - armorFill))
                    esp.ArmorBar.Size = Vector2.new(barW, armorFill)
                    esp.ArmorBar.Color = Color3.fromRGB(0, 210, 255)
                    esp.ArmorBar.Visible = ShowHealthBars

                    -- Armor number text
                    esp.ArmorText.Position = Vector2.new(armorX - 2, boxTop + (barH - armorFill) - 14)
                    esp.ArmorText.Text = tostring(math.floor(armorValue)) .. "/" .. tostring(math.floor(maxArmor))
                    esp.ArmorText.Color = Color3.fromRGB(0, 210, 255)
                    esp.ArmorText.Visible = ShowHealthBars
                else
                    HideAllESP(esp)
                end
            else
                if esp then HideAllESP(esp) end
            end

            -- Aimbot targeting (only alive players)
            if AimbotEnabled and head and playerIsAlive then
                local screenPos, onViewport = Camera:WorldToViewportPoint(head.Position)
                if onViewport then
                    local aimDist = (Vector2.new(screenPos.X, screenPos.Y) - UserInputService:GetMouseLocation()).Magnitude
                    if aimDist < closestDistance then
                        closestDistance = aimDist
                        targetHead = head
                    end
                end
            end
        end

        -- Apply aimbot (framerate-independent lerp + velocity prediction)
        if AimbotEnabled and targetHead then
            local targetChar = targetHead.Parent
            if targetChar then
                local targetHumanoid = targetChar:FindFirstChild("Humanoid")
                if targetHumanoid and targetHumanoid.Health > 0 and targetHumanoid:GetState() ~= Enum.HumanoidStateType.Dead then
                    local safedt = math.clamp(dt or (1/60), 1/240, 1/10)
                    -- Velocity prediction: project head forward by target velocity * dt
                    local predictedPos = targetHead.Position
                    local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
                    if targetHRP then
                        local ok, vel = pcall(function() return targetHRP.AssemblyLinearVelocity end)
                        if not ok then vel = targetHRP.Velocity end
                        predictedPos = targetHead.Position + vel * safedt
                    end
                    -- Framerate-independent lerp: same feel at 30fps, 60fps, 144fps
                    local lerpAlpha = 1 - math.pow(1 - AimbotSmooth, safedt * 60)
                    local goalCFrame = CFrame.lookAt(Camera.CFrame.Position, predictedPos)
                    Camera.CFrame = Camera.CFrame:Lerp(goalCFrame, lerpAlpha)
                end
            end
        end
    end

    local function startFly()
        if not HRP then return end
        FlyEnabled = true
        FlyBodyVelocity = Instance.new("BodyVelocity")
        FlyBodyVelocity.Velocity = Vector3.new(0, 0, 0)
        FlyBodyVelocity.MaxForce = Vector3.new(1e5, 1e5, 1e5)
        FlyBodyVelocity.Parent = HRP
        FlyBodyGyro = Instance.new("BodyGyro")
        FlyBodyGyro.MaxTorque = Vector3.new(1e5, 1e5, 1e5)
        FlyBodyGyro.P = 1e4
        FlyBodyGyro.Parent = HRP

        task.spawn(function()
            while FlyEnabled and HRP do
                local cam = Workspace.CurrentCamera
                local dir = Vector3.new(0, 0, 0)
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + cam.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - cam.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - cam.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + cam.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0, 1, 0) end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.new(0, 1, 0) end
                FlyBodyVelocity.Velocity = dir.Magnitude > 0 and dir.Unit * 50 or Vector3.new(0, 0, 0)
                FlyBodyGyro.CFrame = cam.CFrame
                task.wait()
            end
        end)
    end

    local function stopFly()
        FlyEnabled = false
        if FlyBodyVelocity then FlyBodyVelocity:Destroy(); FlyBodyVelocity = nil end
        if FlyBodyGyro then FlyBodyGyro:Destroy(); FlyBodyGyro = nil end
    end

 
    local function setupGodmode()
        for _, part in pairs(Workspace:GetDescendants()) do
            if part:IsA("BasePart") then
                if string.find(part.Name:lower(), "kill") or string.find(part.Name:lower(), "lava") or part.Material == Enum.Material.Neon then
                    if not storedGodmodeParts[part] then
                        storedGodmodeParts[part] = {
                            CanTouch = part.CanTouch,
                            CanCollide = part.CanCollide,
                            Transparency = part.Transparency
                        }
                    end
                    part.CanTouch = false
                    part.CanCollide = false
                    part.Transparency = 0.5
                end
            end
        end
    end

    local function revertGodmode()
        for part, data in pairs(storedGodmodeParts) do
            if part and part.Parent then
                part.CanTouch = data.CanTouch
                part.CanCollide = data.CanCollide
                part.Transparency = data.Transparency
            end
        end
        storedGodmodeParts = {}
    end

    -- (Wallbang and Trolling functions removed)

    --=============================================
    -- BUILD THE UI MENU
    --=============================================

    -- (Player Targeting section removed)

    -- Multi-Fling
    CreateSection("Multi-Fling Utilities", Color3.fromRGB(255, 150, 0))
    CreateButton("Select Fling Targets", function()
        FlingPanel.Visible = not FlingPanel.Visible
        if FlingPanel.Visible then RefreshFlingList() end
    end)
    CreateToggle("Enable Multi-Fling Loop", function(state)
        FlingActive = state
        if state then
            task.spawn(function()
                while FlingActive do
                    for name, plr in pairs(SelectedTargets) do
                        if not FlingActive then break end
                        if plr and plr.Parent and isAlive(plr) then
                            SkidFling(plr)
                            task.wait(0.1)
                        else
                            SelectedTargets[name] = nil
                        end
                    end
                    task.wait(0.5)
                end
            end)
        end
    end)

    -- Local Player
    CreateSection("Local Player", Color3.fromRGB(0, 170, 255))
    CreateToggle("OP Stats (Invincible & 45 Speed)", function(state)
        OpStatsEnabled = state
        if state then
            if Character then task.spawn(function() ApplyOpStats(Character) end) end
        else
            ClearOpStats()
            if Character then
                local h = Character:FindFirstChildOfClass("Humanoid")
                if h then h.WalkSpeed = 16 end
            end
        end
    end)
    CreateTextBox("Set Speed (default 16)", function(text)
        local s = tonumber(text)
        if s and Character then
            local h = Character:FindFirstChildOfClass("Humanoid")
            if h then h.WalkSpeed = s end
        end
    end)
    CreateTextBox("Set Jump Power (default 50)", function(text)
        local s = tonumber(text)
        if s and Character then
            local h = Character:FindFirstChildOfClass("Humanoid")
            if h then h.JumpPower = s end
        end
    end)
    CreateToggle("Infinite Jump", function(state) InfiniteJump = state end)
    UserInputService.JumpRequest:Connect(function()
        if InfiniteJump and Character then
            local h = Character:FindFirstChildOfClass("Humanoid")
            if h then h:ChangeState(Enum.HumanoidStateType.Jumping) end
        end
    end)
    CreateToggle("No-Clip", function(state) NoClipEnabled = state end)
    RunService.Stepped:Connect(function()
        if NoClipEnabled and Character then
            for _, p in pairs(Character:GetDescendants()) do
                if p:IsA("BasePart") then p.CanCollide = false end
            end
        end
    end)
    CreateToggle("Fly", function(state)
        if state then startFly() else stopFly() end
    end)
    CreateToggle("Invisibility", function(state) EnableInvisibility(state) end)
    CreateToggle("Godmode (Kill Parts Off)", function(state)
        GodmodeEnabled = state
        if state then setupGodmode() else revertGodmode() end
    end)

    -- Aimbot / ESP
    CreateSection("Aimbot / ESP", Color3.fromRGB(255, 255, 0))
    local AimbotToggleFrame = CreateToggle("Enable Aimbot", function(state) AimbotEnabled = state end, "Key", function(txt)
        local key = txt:upper():gsub("%s+", "")
        if key == "" or key == "KEY" then AimbotKeybind = nil; return end
        local ok, kc = pcall(function() return Enum.KeyCode[key] end)
        AimbotKeybind = (ok and kc) and key or nil
    end)
    UserInputService.InputBegan:Connect(function(input, gpe)
        if gpe then return end
        if not AimbotKeybind then return end
        local ok, kc = pcall(function() return Enum.KeyCode[AimbotKeybind] end)
        if ok and kc and input.KeyCode == kc then
            AimbotEnabled = not AimbotEnabled
            -- Sync the visual toggle knob state
            local status = AimbotToggleFrame and AimbotToggleFrame:FindFirstChildOfClass("TextButton")
            local circle = status and status:FindFirstChildOfClass("Frame")
            if status and circle then
                local col = AimbotEnabled and Color3.fromRGB(0, 170, 255) or Color3.fromRGB(15, 15, 15)
                local pos = AimbotEnabled and UDim2.new(1, -20, 0.5, -9) or UDim2.new(0, 2, 0.5, -9)
                TweenService:Create(status, TweenInfo.new(0.2), {BackgroundColor3 = col}):Play()
                TweenService:Create(circle, TweenInfo.new(0.2), {Position = pos}):Play()
            end
        end
    end)
    CreateTextBox("Set Aim Smoothness (0 to 1, default 0.1)", function(txt)
        local v = tonumber(txt)
        if v then AimbotSmooth = math.clamp(v, 0.001, 1) end
    end)

        local v = tonumber(txt)
        if v then ESPMaxDistance = v end
    end)
    
    CreateToggle("Show Health/Armor Bars", function(state) ShowHealthBars = state end)
    
    CreateToggle("Silent Aim", function(state) SilentAimEnabled = state end, "%", function(txt)
        local v = tonumber(txt)
        if v then SilentAimHitChance = math.clamp(v, 0, 100) end
    end)
    
    CreateOptionDropdown("Silent Aim Target", {"Head", "HumanoidRootPart", "Random"}, function(v) SilentAimTargetPart = v end)

    RunService.RenderStepped:Connect(UpdateESP_And_Aimbot)

    --=============================================
    -- Silent Aim Hook
    --=============================================
    local old_namecall
    old_namecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
        local method = getnamecallmethod()
        if SilentAimEnabled and (method == "FireServer" or method == "InvokeServer") then
            local args = {...}
            if math.random(1, 100) <= SilentAimHitChance then
                local target = GetSilentAimTarget()
                if target then
                    for i, v in pairs(args) do
                        if typeof(v) == "Vector3" then args[i] = target.Position
                        elseif typeof(v) == "CFrame" then args[i] = target.CFrame end
                    end
                    return old_namecall(self, unpack(args))
                end
            end
        end
        return old_namecall(self, ...)
    end))
